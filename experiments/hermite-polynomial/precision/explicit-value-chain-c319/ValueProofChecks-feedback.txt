def HermiteExplicitCachedStored.comparisonQ : ℕ → ℕ → Fin 2 → Fin 3 → Fin 3 → StoredBinaryCoordinates.IndexedRun ℚ :=
fun cut q bit a b =>
  have digit := cut / 2 ^ q % 2;
  have low := charge Op.compare (decide (↑bit < digit));
  have high := charge Op.compare (decide (digit < ↑bit));
  have next := if low.value = true then 0 else if high.value = true then 2 else a;
  have hit := charge Op.compare (decide (next = b));
  { run := { value := if hit.value = true then 1 else 0, cost := low.cost + high.cost + hit.cost }, quotientCalls := 1,
    remainderCalls := 1 }
