def HermiteExplicitCachedStored.frontierBudget : ℕ → ℕ → ℚ → ℕ :=
fun k n delta =>
  have T := HermiteFiniteExp.tailCutoff delta;
  have d := HermiteFiniteExpDegree.sourceDegree delta;
  have L := d + 2 * k + 1;
  have D := HermitePiecewiseAssembly.assemblySize k delta;
  have P := (n + 1) * (2 * D ^ 2 + D + 1) + 2 * D;
  816 * (k + 1) ^ 2 + 81 * (d + 1) ^ 2 + 8 * (d + 1) + 80 * (L + 1) ^ 2 + 2 * (n + 1) * D ^ 2 * (n + L + 34) +
                D * (L + 61) +
              9 * P +
            20 * D ^ 2 +
          5 * n ^ 2 +
        9 * n +
      T +
    59
def HermitePiecewiseSourceBudget.sourceDelta : ℕ → ℕ → ℚ → ℚ :=
fun k n epsilon => epsilon / (4 * 2 ^ (n + 1) * HermiteFiniteMiddleSource.amplificationQ k)
def HermiteFiniteExpDegree.sourceDegree : ℚ → ℕ :=

def HermiteFiniteExp.tailCutoff : ℚ → ℕ :=
fun epsilon => Nat.clog 2 (max 1 ⌈1 / epsilon⌉₊)
