def HermiteSourceSuffix.ActualSuffixSpec : Prop :=
∀ (k : ℕ) (i : Fin (k + 1)) (s : Fin (↑i + 2)),
  actualSuffixObservation k i s = SourceSuffix k i s ∧
    actualSuffixObservation k i ⟨0, ⋯⟩ = (HermiteChargedSourceCache.sourceEntry k (actualFactorials k) i).value ∧
      if h : ↑s < ↑i + 1 then
        have next := ⟨↑s + 1, ⋯⟩;
        actualTailObservation k i s = actualSuffixObservation k i next ∧
          actualSuffixObservation k i s = (actualSourceTerm k i ⟨↑s, h⟩).value + actualTailObservation k i s
      else actualSuffixObservation k i s = 0
def HermiteSourceSuffix.SuffixEnvelopeSpec : Prop :=
∀ (k : ℕ) (i : Fin (k + 1)) (s : Fin (↑i + 2)),
  have q := SourceSuffix k i s;
  q.den ∣ k.factorial ∧ 0 ≤ q ∧ q ≤ ↑((↑i + 1 - ↑s) * 2 ^ (2 * k))
